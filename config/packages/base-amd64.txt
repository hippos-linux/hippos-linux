# amd64-specific base packages

# Processor microcode
intel-microcode
amd64-microcode

# GRUB EFI bootloader for x86_64
grub-efi-amd64

# Intel Xorg DDX driver (supplements modesetting for Gen9- hardware)
xserver-xorg-video-intel

# Intel VA-API (free tier; non-free tier installed by configure-rootfs.sh)
intel-media-va-driver

# i386 runtime libraries required by linuxloader (linuxloader.so)
libsdl3-0:i386
libsdl3-image0:i386
libsdl3-ttf0:i386
libglut3.12:i386

# Intel thermal daemon (x86 only)
thermald

# Sinden light gun software (downloaded from Sinden on first use) runs on Mono
mono-runtime
libmono-system4.0-cil
libmono-system-core4.0-cil
libmono-system-configuration4.0-cil
libmono-system-drawing4.0-cil
libgdiplus
libsdl1.2-compat
